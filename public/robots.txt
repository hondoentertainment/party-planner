# Legal (public)

# Authenticated app surfaces (HTML is still SPA-delivered; this signals intent to crawlers.)
Disallow: /events/

# Carries a single-use HMAC token in the query string (notify-unsubscribe redirects
# here on click). Crawling/archiving these would log the token in third-party caches.
Disallow: /email/unsubscribe

# Public guest invites: the index page is shareable, but the recovery query
# parameter (?rsvp_token=...) is private to the recipient. Crawlers should
# index the bare /s/<token> page only.
Allow: /s/
Disallow: /s/*?*rsvp_token=*
